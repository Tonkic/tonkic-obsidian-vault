-- 1. 创建原始表并加载清洗后的数据 (注意路径要和 WorkflowTest 输出一致)
CREATE TABLE IF NOT EXISTS ods_jobdata_origin(city string,salary array<string>,company array<string>,kill array<string>)

CREATE TABLE IF NOT EXISTS t_ods_city(every_city string, counts int) ROW FORMAT DELIMITED FIELDS TERMINATED BY ',' STORED AS TEXTFILE;
INSERT OVERWRITE TABLE t_ods_city SELECT city, count(1) FROM ods_jobdata_origin GROUP BY city;

-- 3. 统计技能数据 (t_ods_kill)
-- 先炸裂数组
CREATE TABLE IF NOT EXISTS t_ods_tmp_kill AS SELECT explode(kill) FROM ods_jobdata_origin;
-- 再统计
CREATE TABLE IF NOT EXISTS t_ods_kill(every_kill string, counts int) ROW FORMAT DELIMITED FIELDS TERMINATED BY ',' STORED AS TEXTFILE;
INSERT OVERWRITE TABLE t_ods_kill SELECT col, count(1) FROM t_ods_tmp_kill GROUP BY col;

-- 4. 统计福利数据 (t_ods_company)
CREATE TABLE IF NOT EXISTS t_ods_tmp_company AS SELECT explode(company) FROM ods_jobdata_origin;
CREATE TABLE IF NOT EXISTS t_ods_company(every_company string, counts int) ROW FORMAT DELIMITED FIELDS TERMINATED BY ',' STORED AS TEXTFILE;
INSERT OVERWRITE TABLE t_ods_company SELECT col, count(1) FROM t_ods_tmp_company GROUP BY col;

-- 5. 统计薪资数据 (t_ods_salary) - 简化逻辑
-- 实际项目可能需要复杂的 case when 区间分组，这里为了演示流程简化为直接统计

CREATE TABLE IF NOT EXISTS t_ods_salary(every_partition string, counts int) ROW FORMAT DELIMITED FIELDS TERMINATED BY ',' STORED AS TEXTFILE;
