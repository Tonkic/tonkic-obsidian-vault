##### 在bigdata\src\test\java\cn\edu\scnu下新建HDFSDemo1.java（AppTest.java同目录）
```java
package cn.edu.scnu; 

import java.io.FileOutputStream;
import java.io.IOException;
import java.io.InputStream; 
import java.net.URI;
import java.nio.charset.StandardCharsets;

import org.apache.hadoop.conf.Configuration;
import org.apache.hadoop.fs.*;
import org.apache.hadoop.io.IOUtils;
import org.junit.After;
import org.junit.Before;
import org.junit.Test;

public class HDFSDemo1 {

    private FileSystem fs;
    // --- 请根据您的环境确认以下配置 ---
    private final String HDFS_URI = "hdfs://192.168.243.128:9000"; // 已修正为您集群的IP地址
    private final String HDFS_USER = "hadoop";

    /**
     * @Before 注解会在每个 @Test 方法运行前执行
     * 用于初始化 FileSystem 对象，避免代码重复
     */
    @Before
    public void setup() throws IOException, InterruptedException {
        Configuration conf = new Configuration();
        fs = FileSystem.get(URI.create(HDFS_URI), conf, HDFS_USER);
        System.out.println("--- HDFS Connection Initialized ---");
    }

    /**
     * @After 注解会在每个 @Test 方法运行后执行
     * 用于关闭 FileSystem 连接，确保资源释放
     */
    @After
    public void tearDown() throws IOException {
        if (fs != null) {
            fs.close();
            System.out.println("--- HDFS Connection Closed ---");
        }
    }

    /**
     * 1. 在HDFS中创建文件并写入内容
     */
    @Test
    public void createFile() throws IOException {
        Path path = new Path("/input.txt");
        String content = "hello world & 世界您好！";
        // 使用 try-with-resources 自动关闭输出流
        try (FSDataOutputStream fsout = fs.create(path, true)) { // true 表示如果文件已存在则覆盖
            fsout.write(content.getBytes(StandardCharsets.UTF_8));
        }
        System.out.println("文件创建成功！路径: " + path);
    }

    /**
     * 2. 读取文件内容并打印到控制台
     */
    @Test
    public void readFile() throws IOException {
        Path path = new Path("/input.txt");
        if (!fs.exists(path)) {
            System.out.println("文件 " + path + " 不存在！");
            return;
        }
        System.out.println("正在读取文件: " + path);
        // 使用 try-with-resources 自动关闭输入流
        try (InputStream in = fs.open(path)) {
            IOUtils.copyBytes(in, System.out, 4096, false); // false 表示不关闭 System.out
        }
    }

    /**
     * 3. 从本地上传文件到HDFS
     */
    @Test
    public void uploadFile() throws IOException {
        // 确保你的项目根目录下有一个名为 "b.txt" 的文件
        // 如果没有，可以先运行下面的 createFileOnLocalSystem() 测试方法来创建一个
        Path localPath = new Path("b.txt");
        Path hdfsPath = new Path("/b.txt");
        
        fs.copyFromLocalFile(false, true, localPath, hdfsPath); // false=不删除源文件, true=覆盖目标文件
        
        System.out.println("文件上传成功！");
        System.out.println("本地路径: " + localPath);
        System.out.println("HDFS路径: " + hdfsPath);
        
        // 验证文件是否上传成功
        if (fs.exists(hdfsPath)) {
            System.out.println("验证成功：HDFS中已存在文件 " + hdfsPath);
        } else {
            System.out.println("验证失败：文件上传可能未成功。");
        }
    }

    /**
     * 4. 从HDFS下载文件到本地
     */
    @Test
    public void downloadFile() throws IOException {
        Path hdfsPath = new Path("/input.txt");
        Path localPath = new Path("downloaded_from_hdfs.txt");
        
        if (!fs.exists(hdfsPath)) {
            System.out.println("HDFS文件 " + hdfsPath + " 不存在，无法下载！");
            return;
        }

        fs.copyToLocalFile(false, hdfsPath, localPath, true); // true表示使用本地文件系统的校验和
        
        System.out.println("文件下载成功！");
        System.out.println("HDFS路径: " + hdfsPath);
        System.out.println("本地路径: " + localPath);
    }

    /**
     * (辅助功能) 在本地项目根目录创建一个用于上传测试的文件
     */
    @Test
    public void createFileOnLocalSystem() throws IOException {
        try (FileOutputStream fos = new FileOutputStream("local_file_to_upload.txt")) {
            fos.write("This is a local file for upload test.".getBytes(StandardCharsets.UTF_8));
        }
        System.out.println("本地测试文件创建成功: local_file_to_upload.txt");
    }
}
```
