```bash
_ssh-keygen -t rsa -P '' -f ~/.ssh/id_rsa_

# 到Slave03后复制回前面四个

scp authorized_keys root@Master02:/root/.ssh/
scp authorized_keys root@Slave01:/root/.ssh/
scp authorized_keys root@Slave02:/root/.ssh/
```
