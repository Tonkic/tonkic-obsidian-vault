tmux kill-session -t cli

##### update.sh
```bash
#!/usr/bin/env bash
set -Eeuo pipefail

REPO="router-for-me/CLIProxyAPIPlus"
INSTALL_DIR="/root/CLIProxyAPIPlus"
BIN_PATH="$INSTALL_DIR/bin/cli-proxy-api-plus"
CONFIG_PATH="$INSTALL_DIR/config.yaml"
BACKUP_DIR="$INSTALL_DIR/backup"
TMP_DIR="$INSTALL_DIR/tmp"
LOG_DIR="$INSTALL_DIR/logs"
STATE_FILE="$INSTALL_DIR/current_version.txt"
TMUX_SESSION="cli"
LOCK_FILE="/tmp/cliproxy_update.lock"

mkdir -p "$BACKUP_DIR" "$TMP_DIR" "$LOG_DIR"

exec 9>"$LOCK_FILE"
flock -n 9 || {
  echo "已有更新任务在运行，退出。"
  exit 0
}

log() {
  echo "[$(date '+%F %T')] $*"
}

cleanup() {
  rm -rf "$TMP_DIR"/*
}
trap cleanup EXIT

API_URL="https://api.github.com/repos/$REPO/releases/latest"

log "读取 GitHub 最新 release 信息..."
JSON="$(curl -fsSL \
  -H 'Accept: application/vnd.github+json' \
  -H 'X-GitHub-Api-Version: 2022-11-28' \
  "$API_URL")"

LATEST_TAG="$(printf '%s' "$JSON" | grep -m1 '"tag_name":' | sed -E 's/.*"([^"]+)".*/\1/')"

if [[ -z "${LATEST_TAG:-}" ]]; then
  log "无法解析最新版本号"
  exit 1
fi

ASSET_URL="$(printf '%s' "$JSON" \
  | grep 'browser_download_url' \
  | grep 'linux_amd64.tar.gz' \
  | head -n1 \
  | cut -d '"' -f4)"

if [[ -z "${ASSET_URL:-}" ]]; then
  log "未找到 linux_amd64.tar.gz 下载地址"
  exit 1
fi

if [[ -f "$STATE_FILE" ]]; then
  CURRENT_TAG="$(cat "$STATE_FILE")"
else
  CURRENT_TAG="$("$BIN_PATH" --version 2>/dev/null | head -n1 | grep -oE 'v?[0-9]+\.[0-9]+\.[0-9]+-[0-9]+' || true)"
fi

log "当前版本: ${CURRENT_TAG:-unknown}"
log "最新版本: $LATEST_TAG"

if [[ "${CURRENT_TAG:-}" == "$LATEST_TAG" ]]; then
  log "已是最新版本，无需更新。"
  exit 0
fi

ARCHIVE_NAME="$TMP_DIR/${LATEST_TAG}_linux_amd64.tar.gz"
EXTRACT_DIR="$TMP_DIR/extract"
mkdir -p "$EXTRACT_DIR"

log "下载新版本..."
curl -fL "$ASSET_URL" -o "$ARCHIVE_NAME"

log "解压缩..."
tar -xzf "$ARCHIVE_NAME" -C "$EXTRACT_DIR"

NEW_BIN="$(find "$EXTRACT_DIR" -type f -name 'cli-proxy-api-plus' | head -n1)"
if [[ -z "${NEW_BIN:-}" ]]; then
  log "解压后未找到 cli-proxy-api-plus"
  exit 1
fi

chmod +x "$NEW_BIN"

TIMESTAMP="$(date '+%Y%m%d_%H%M%S')"
if [[ -f "$BIN_PATH" ]]; then
  cp -a "$BIN_PATH" "$BACKUP_DIR/cli-proxy-api-plus.${CURRENT_TAG:-unknown}.${TIMESTAMP}.bak"
  log "已备份旧二进制"
fi

if tmux has-session -t "$TMUX_SESSION" 2>/dev/null; then
  log "关闭旧 tmux 会话: $TMUX_SESSION"
  tmux kill-session -t "$TMUX_SESSION"
else
  log "旧 tmux 会话不存在，跳过关闭"
fi

log "替换二进制..."
install -m 755 "$NEW_BIN" "$BIN_PATH"
echo "$LATEST_TAG" > "$STATE_FILE"

log "启动新 tmux 会话..."
tmux new-session -d -s "$TMUX_SESSION" "cd '$INSTALL_DIR' && '$BIN_PATH' -config '$CONFIG_PATH' >> '$LOG_DIR/runtime.log' 2>&1"

sleep 3

if tmux has-session -t "$TMUX_SESSION" 2>/dev/null; then
  log "tmux 会话已启动: $TMUX_SESSION"
else
  log "新 tmux 会话启动失败，开始回滚..."
  LAST_BACKUP="$(ls -1t "$BACKUP_DIR"/cli-proxy-api-plus.*.bak 2>/dev/null | head -n1 || true)"
  if [[ -n "${LAST_BACKUP:-}" ]]; then
    install -m 755 "$LAST_BACKUP" "$BIN_PATH"
    tmux new-session -d -s "$TMUX_SESSION" "cd '$INSTALL_DIR' && '$BIN_PATH' -config '$CONFIG_PATH' >> '$LOG_DIR/runtime.log' 2>&1"
    log "已回滚并尝试重新启动旧版本"
  else
    log "没有可用备份，需手动处理"
  fi
  exit 1
fi

log "更新完成，当前版本: $LATEST_TAG"
```
